CREATE TABLE Auto (
rendszam CHAR(7) PRIMARY KEY,
marka VARCHAR(50) NOT NULL,
tipus VARCHAR(50) NOT NULL,
napi_dij INT CHECK (napi_dij > 0)
);

CREATE TABLE Ugyfel (
ugyfel_id INT PRIMARY KEY,
nev VARCHAR(80) NOT NULL,
telefon VARCHAR(20),
email VARCHAR(100) NOT NULL UNIQUE KEY,
);


CREATE TABLE Kolcsonzes (
kolcsonzes_id INT PRIMARY KEY,
auto_rendszam CHAR(7) FORGEIN KEY (auto_rendszam) REFERENCES Auto(rendszam) NOT NULL,
ugyfel_id INT FORGEIN KEY (ugyfel_id) REFERENCES Ugyfel(ugyfel_id) NOT NULL,
napok_szama INT CHECK (napok_szama > 0)
);
